C++ Native Windows SDK

📥 Download JoystAuth.hpp 📄 Sample main.cpp
📦 Integration Steps:
  1. Download JoystAuth.hpp and paste it into your Visual Studio project directory.
  2. Include #include "JoystAuth.hpp" at the top of your code.
  3. Header-only with zero external dependencies! (WinINet is automatically linked).
Implementation Example (main.cpp):
#include "JoystAuth.hpp"
#include <iostream>

int main() {
    // 1. Initialize with your App Name and Master App Token
    JoystAuth::api auth("MyApp", "YOUR_APP_TOKEN");

    // 2. Run Dynamic Handshake
    if (!auth.init()) {
        std::cout << "Initialization failed: " << auth.response.message << "\n";
        return 1;
    }

    // 3. User Login with Motherboard HWID Lock
    if (auth.login("gamer123", "password123")) {
        std::cout << "✅ " << auth.response.message << "\n";
        std::cout << "Subscription Tier: " << auth.user_data.subscription << "\n";
    } else {
        std::cout << "❌ " << auth.response.message << "\n";
    }

    return 0;
}

MEMORY SAFE & ZERO COST ASYNC

🦀 Rust Async Tokio SDK

📥 Download main.rs
📦 Integration Steps:
  1. Add tokio and reqwest to your Cargo.toml.
  2. Include the client implementation from src/main.rs in your crate.
#[tokio::main]
async fn main() -> Result<(), Box<dyn std::error::Error>> {
    let mut auth = JoystAuth::new("MyApp", "YOUR_APP_TOKEN", "1.0", "https://joystauth.cc");
    
    // Handshake
    auth.init().await?;
    
    // Authenticate
    if auth.login("gamer123", "password123").await? {
        println!("✅ Logged in as: {}", auth.user_data.username);
    } else {
        println!("❌ Login failed: {}", auth.response.message);
    }
    Ok(())
}

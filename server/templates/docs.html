C++ Native Windows SDK

📥 Download AuthClient.hpp 📄 Sample main.cpp
📦 Integration Steps:
  1. Download AuthClient.hpp and paste it into your Visual Studio project directory.
  2. Include #include "AuthClient.hpp" at the top of your code.
  3. Link wininet.lib under Project Properties > Linker > Input > Additional Dependencies.
Implementation Example (main.cpp):
#include "AuthClient.hpp"
#include <iostream>

int main() {
    // 1. Initialize with your App Name and Master App Token
    JoystAuth::api auth(
        "MyApp",              // Application Name
        "sec_77918750...",    // Master App Token from Dashboard
        "1.0"                 // Version
    );

    // 2. Run Anti-Debug & Dynamic AES-256 Handshake
    if (!auth.init()) {
        std::cout << "Initialization failed: " << auth.response.message << "\n";
        return 1;
    }

    // 3. User Login with Motherboard HWID Lock
    if (auth.login("gamer123", "password123")) {
        std::cout << "Login Successful! Welcome " << auth.user_data.username << "\n";
        std::cout << "Subscription Tier: " << auth.user_data.subscription << "\n";
        std::cout << "Expiry Date: " << auth.user_data.expiry << "\n";

        // 4. Start 30-second Session Security Heartbeat Thread
        auth.start_heartbeat(30);

        // 5. Fetch Secure Encrypted Cloud Variable
        std::string offset = auth.var("aimbot_offset");
        std::cout << "Secure Offset: " << offset << "\n";
    } else {
        std::cout << "Access Denied: " << auth.response.message << "\n";
    }

    return 0;
}

public class Main {
    public static void main(String[] args) {
        // 1. Initialize API Instance

            System.err.println("Init failed: " + auth.response.message);
            return;

}

Rust Client SDK (Cargo)

📥 Download main.rs 📄 Cargo.toml
📦 Integration Steps:
  1. Add tokio = { version = "1.0", features = ["full"] } and reqwest = { version = "0.11", features = ["json"] } to your Cargo.toml.
  2. Copy sample main.rs to your src/ folder and run cargo run.
use joyst_auth::JoystApi;

#[tokio::main]
async fn main() -> Result<(), Box<dyn std::error::Error>> {
    // 1. Initialize API Client
    let mut auth = JoystApi::new("MyApp", "sec_77918750...", "1.0");
    auth.init().await?;

    // 2. Login with HWID Binding
    if auth.login("gamer123", "password123").await? {
        println!("User Authenticated: {}", auth.user.username);
        println!("Rank: {}", auth.user.subscription);
        
        // 3. Start Tokio Heartbeat Task
        auth.start_heartbeat(30);
    } else {
        println!("Login Failed: {}", auth.last_error);
    }

    Ok(())
}
